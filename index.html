01 / THE TECHNOLOGY

Engineered
to drive.

Discover the technology behind your Emoon3D RC car.

01 / CONTROLLER

ESP32 brain

The ESP32 is the car's main controller. It creates its own Wi-Fi network, so your phone can connect directly without mobile data or a separate router.

02 / CONNECTION

Wi-Fi + WebSocket

Connect your phone to the car and open its control page. Live wireless commands keep the steering and throttle responding as you drive.

03 / DRIVE

Motor driver

A TB6612FNG motor driver controls the drive motor. The ESP32 sets speed with PWM (GPIO 25) and direction with GPIO 26 and 27. The driver is switched off through the standby pin (GPIO 33) whenever the car is stopped.

04 / STEERING

Servo control

An SG90 servo steers the car. The firmware moves it between 0° and 40°, with the centre at 20°. The servo signal is released after 0.3 seconds without steering changes.

05 / MOTOR RESPONSE

Smooth acceleration

The motor ramps up gradually instead of jumping to full power, and the maximum motor power is capped at 200 out of 255.

06 / SAFETY

Automatic stop

If the phone stops sending driving commands, the motor automatically stops after about 0.6 seconds. Always use the physical power switch when you finish driving.

#include <WiFi.h>
#include <WebServer.h>
#include <WebSocketsServer.h>
#include "soc/soc.h"
#include "soc/rtc_cntl_reg.h"

// ===================================
// TEST / TUNING SWITCHES
// ===================================

#define DISABLE_BROWNOUT true     // true = ESP32 survives short voltage dips
#define SERVO_ENABLED    true     // set false to test if clicking comes from servo

const int MOTOR_MAX_PWM = 200;    // 0-255, lower = less current
const int RAMP_UP_STEP = 6;       // PWM added every RAMP_INTERVAL (soft start)
const int RAMP_DOWN_STEP = 20;    // faster slow-down
const unsigned long RAMP_INTERVAL = 10;  // ms

// ===================================
// WIFI SETTINGS
// ===================================

const char* ssid = "MY_RC_CAR";
const char* password = "12345678";

WebServer server(80);
WebSocketsServer webSocket(81);

// ===================================
// ESP32 PIN CONNECTIONS (UNCHANGED)
// ===================================

#define MOTOR_PWM 25
#define MOTOR_IN1 26
#define MOTOR_IN2 27
#define MOTOR_STBY 33

#define SERVO_PIN 18

// ===================================
// MOTOR SETTINGS
// ===================================

const int PWM_FREQ = 20000;
const int PWM_RESOLUTION = 8;
const int THROTTLE_DEADZONE = 25;

// ===================================
// SERVO SETTINGS
// ===================================

const int SERVO_FREQ = 50;
const int SERVO_RESOLUTION = 14;

const int STEER_MIN_ANGLE = 0;
const int STEER_MAX_ANGLE = 40;
const int STEER_CENTER = 20;

const int STEER_PULSE_MIN = 1200;   // at 0 deg
const int STEER_PULSE_MAX = 1800;   // at 40 deg

const unsigned long SERVO_IDLE_RELEASE = 300;

// ===================================
// CONTROL VARIABLES
// ===================================

int targetThrottle = 0;    // what the user asks for
int currentThrottle = 0;   // what the motor is actually doing (ramped)
int steering = STEER_CENTER;

bool driverOn = false;
bool servoActive = false;

unsigned long lastCommandTime = 0;
unsigned long lastSteerChange = 0;
unsigned long lastRamp = 0;

const unsigned long FAILSAFE_TIME = 600;

// ===================================
// SERVO CONTROL
// ===================================

void setSteering(int angle) {

  if (!SERVO_ENABLED) return;

  angle = constrain(angle, STEER_MIN_ANGLE, STEER_MAX_ANGLE);

  int pulse = map(angle, STEER_MIN_ANGLE, STEER_MAX_ANGLE,
                  STEER_PULSE_MIN, STEER_PULSE_MAX);

  int duty = (pulse * 16384L) / 20000;

  ledcWrite(SERVO_PIN, duty);

  servoActive = true;
  lastSteerChange = millis();
}

void releaseServo() {
  ledcWrite(SERVO_PIN, 0);
  servoActive = false;
}

// ===================================
// MOTOR CONTROL
// ===================================

void driverEnable(bool on) {
  if (on == driverOn) return;
  driverOn = on;
  digitalWrite(MOTOR_STBY, on ? HIGH : LOW);
}

void motorOff() {
  ledcWrite(MOTOR_PWM, 0);
  digitalWrite(MOTOR_IN1, LOW);
  digitalWrite(MOTOR_IN2, LOW);
  driverEnable(false);
}

void applyMotor(int speed) {

  speed = constrain(speed, -255, 255);

  if (abs(speed) < THROTTLE_DEADZONE) {
    motorOff();
    return;
  }

  driverEnable(true);

  if (speed > 0) {
    digitalWrite(MOTOR_IN1, HIGH);
    digitalWrite(MOTOR_IN2, LOW);
  } else {
    digitalWrite(MOTOR_IN1, LOW);
    digitalWrite(MOTOR_IN2, HIGH);
  }

  ledcWrite(MOTOR_PWM, abs(speed));
}

// Soft start / soft stop: called from loop()
void updateMotorRamp() {

  if (millis() - lastRamp < RAMP_INTERVAL) return;
  lastRamp = millis();

  // limit max power
  int target = constrain(targetThrottle, -MOTOR_MAX_PWM, MOTOR_MAX_PWM);

  if (currentThrottle == target) return;

  // slowing down or reversing -> faster step, speeding up -> gentle step
  bool speedingUp = abs(target) > abs(currentThrottle) &&
                    (target >= 0) == (currentThrottle >= 0);
  int step = speedingUp ? RAMP_UP_STEP : RAMP_DOWN_STEP;

  if (currentThrottle < target) {
    currentThrottle = min(currentThrottle + step, target);
  } else {
    currentThrottle = max(currentThrottle - step, target);
  }

  applyMotor(currentThrottle);
}

void stopCar() {
  targetThrottle = 0;
  currentThrottle = 0;
  motorOff();
}

// ===================================
// WEBSOCKET EVENTS
// ===================================

void onWebSocketEvent(uint8_t num, WStype_t type, uint8_t* payload, size_t length) {

  if (type == WStype_TEXT) {

    int s, t;

    if (sscanf((char*)payload, "%d,%d", &s, &t) == 2) {

      lastCommandTime = millis();

      int newSteer = constrain(s, STEER_MIN_ANGLE, STEER_MAX_ANGLE);
      if (newSteer != steering) {
        steering = newSteer;
        setSteering(steering);
      }

      targetThrottle = constrain(t, -255, 255);
    }
  }
  else if (type == WStype_DISCONNECTED) {
    stopCar();
    steering = STEER_CENTER;
    setSteering(STEER_CENTER);
  }
}

void handleRoot() {
  server.send_P(200, "text/html", webpage);
}

// ===================================
// RESET REASON (for diagnosis)
// ===================================

void printResetReason() {
  esp_reset_reason_t r = esp_reset_reason();
  Serial.print("Reset reason: ");
  switch (r) {
    case ESP_RST_POWERON:  Serial.println("POWER ON (normal)"); break;
    case ESP_RST_BROWNOUT: Serial.println("BROWNOUT (voltage dip!)"); break;
    case ESP_RST_PANIC:    Serial.println("PANIC (code crash)"); break;
    case ESP_RST_SW:       Serial.println("SOFTWARE RESET"); break;
    case ESP_RST_WDT:
    case ESP_RST_INT_WDT:
    case ESP_RST_TASK_WDT: Serial.println("WATCHDOG"); break;
    default:               Serial.println((int)r); break;
  }
}

// ===================================
// SETUP
// ===================================

void setup() {

  // Motor driver OFF immediately
  pinMode(MOTOR_STBY, OUTPUT);
  digitalWrite(MOTOR_STBY, LOW);

  pinMode(MOTOR_IN1, OUTPUT);
  pinMode(MOTOR_IN2, OUTPUT);
  digitalWrite(MOTOR_IN1, LOW);
  digitalWrite(MOTOR_IN2, LOW);

  pinMode(MOTOR_PWM, OUTPUT);
  digitalWrite(MOTOR_PWM, LOW);

  if (DISABLE_BROWNOUT) {
    WRITE_PERI_REG(RTC_CNTL_BROWN_OUT_REG, 0);
  }

  Serial.begin(115200);
  delay(100);
  Serial.println();
  printResetReason();

  ledcAttach(MOTOR_PWM, PWM_FREQ, PWM_RESOLUTION);
  ledcWrite(MOTOR_PWM, 0);

  ledcAttach(SERVO_PIN, SERVO_FREQ, SERVO_RESOLUTION);
  ledcWrite(SERVO_PIN, 0);

  setSteering(STEER_CENTER);

  // Wi-Fi: lower power = smaller current peaks
  WiFi.mode(WIFI_AP);
  WiFi.setSleep(false);
  WiFi.softAP(ssid, password, 1, 0, 2);
  WiFi.setTxPower(WIFI_POWER_15dBm);

  Serial.println("RC CAR STARTED");
  Serial.print("Control URL: http://");
  Serial.println(WiFi.softAPIP());

  server.on("/", handleRoot);
  server.begin();

  webSocket.begin();
  webSocket.onEvent(onWebSocketEvent);

  lastCommandTime = millis();
}

// ===================================
// MAIN LOOP
// ===================================

void loop() {

  webSocket.loop();
  server.handleClient();

  // Failsafe: no command -> stop
  if (targetThrottle != 0 && millis() - lastCommandTime > FAILSAFE_TIME) {
    stopCar();
  }

  updateMotorRamp();

  // Release servo signal when steering is still
  if (servoActive && millis() - lastSteerChange > SERVO_IDLE_RELEASE) {
    releaseServo();
  }
}

07 / QUICK HELP

Before first
power-on.

How the current firmware behaves.

What happens when I connect the battery?

When the battery is connected, the ESP32 starts, switches the motor driver off, moves the steering servo once to its 20° centre position and starts the MY_RC_CAR Wi-Fi network. The motor stays off until you move the throttle joystick.

How do I connect my phone?

Join the Wi-Fi network named MY_RC_CAR using the password configured in the sketch (12345678). Open the Control URL printed in Serial Monitor. The default address is 192.168.4.1. The exact address is also printed in Serial Monitor.

What happens if the phone disconnects?

On WebSocket disconnect, the firmware calls stopCar() and returns steering to centre. While driving, if no fresh command arrives for 600 milliseconds, the failsafe stops the motor.

Which pins does the car use?

GPIO 25 (motor PWM), GPIO 26 (IN1), GPIO 27 (IN2), GPIO 33 (STBY) and GPIO 18 (servo signal).
